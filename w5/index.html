Google Apps Script API 設定

若要進行真正的跨裝置多人在線連線，請輸入部署好的 Google Apps Script 網頁應用程式 URL (網址)。若留空則會自動使用「本地模擬/多頁頁籤模擬模式」。

Google Apps Script (GAS) 後端腳本

請在 Google Sheet 的「擴充功能 > Apps Script」中貼上以下程式碼，並部署為「網頁應用程式 (Web App)」，存取權限設定為「所有人 (Anyone)」。試算表分頁名稱將自動建立為 TelepathyData。 


const SHEET_NAME = 'TelepathyData';

function doGet(e) {
  const roomCode = e.parameter.roomCode;
  const action = e.parameter.action;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['RoomCode', 'Question', 'IsRevealed', 'AnswersJSON', 'UpdatedAt']);
  }
  
  if (action === 'get') {
    const data = getRoomData(sheet, roomCode);
    return ContentService.createTextOutput(JSON.stringify(data))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  const params = JSON.parse(e.postData.contents);
  const action = params.action;
  const roomCode = params.roomCode;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['RoomCode', 'Question', 'IsRevealed', 'AnswersJSON', 'UpdatedAt']);
  }

  let roomData = getRoomData(sheet, roomCode) || {
    question: "等待房主出題中...",
    isRevealed: false,
    answers: {}
  };

  if (action === 'publishQuestion') {
    roomData.question = params.question;
    roomData.isRevealed = false;
    roomData.answers = {}; // 清空前一輪回答
  } else if (action === 'submitAnswer') {
    roomData.answers[params.userName] = params.answer;
  } else if (action === 'triggerReveal') {
    roomData.isRevealed = true;
  } else if (action === 'resetAnswers') {
    roomData.answers = {};
    roomData.isRevealed = false;
  } else if (action === 'clearRoom') {
    roomData = { question: "等待房主出題中...", isRevealed: false, answers: {} };
  }

  saveRoomData(sheet, roomCode, roomData);

  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: roomData }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getRoomData(sheet, roomCode) {
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(roomCode)) {
      return {
        question: rows[i][1],
        isRevealed: rows[i][2] === true || rows[i][2] === 'true',
        answers: JSON.parse(rows[i][3] || '{}')
      };
    }
  }
  return null;
}

function saveRoomData(sheet, roomCode, data) {
  const rows = sheet.getDataRange().getValues();
  let rowIndex = -1;
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(roomCode)) {
      rowIndex = i + 1;
      break;
    }
  }
  
  const jsonAnswers = JSON.stringify(data.answers || {});
  const now = new Date();
  
  if (rowIndex > 0) {
    sheet.getRange(rowIndex, 2, 1, 4).setValues([[data.question, data.isRevealed, jsonAnswers, now]]);
  } else {
    sheet.appendRow([roomCode, data.question, data.isRevealed, jsonAnswers, now]);
  }
}
