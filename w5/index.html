對應後端 Google Apps Script (Code.gs) 範本 
const SHEET_NAME = 'TelepathyData';

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow(['roomId', 'round', 'status', 'question', 'playerId', 'playerName', 'answer', 'isHost', 'updatedAt']);
    }

    const params = e.parameter || {};
    let payload = {};
    if (e.postData && e.postData.contents) {
      payload = JSON.parse(e.postData.contents);
    }
    const action = payload.action || params.action || 'getRoom';
    const roomId = (payload.roomId || params.roomId || '').toUpperCase();

    const data = sheet.getDataRange().getValues();
    const headers = data[0];

    if (action === 'getRoom') {
      const rows = [];
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).toUpperCase() === roomId) {
          rows.push({
            roomId: data[i][0], round: Number(data[i][1]), status: data[i][2],
            question: data[i][3], playerId: data[i][4], playerName: data[i][5],
            answer: data[i][6], isHost: Boolean(data[i][7]), updatedAt: data[i][8]
          });
        }
      }
      return jsonResponse({ ok: true, rows: rows });
    }

    if (action === 'upsertPlayer') {
      const now = new Date().toISOString();
      let foundRow = -1;
      let currentRound = 1, currentStatus = 'WAITING', currentQuestion = '等待房主出題...';
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).toUpperCase() === roomId) {
          currentRound = Number(data[i][1]) || 1;
          currentStatus = data[i][2] || 'WAITING';
          currentQuestion = data[i][3] || currentQuestion;
          if (String(data[i][4]) === String(payload.playerId)) foundRow = i + 1;
        }
      }
      if (payload.isNewRoom) {
        currentQuestion = payload.question || '說到夏天第一個想到的食物？';
      }
      const rowValues = [
        roomId, currentRound, currentStatus, currentQuestion,
        payload.playerId, payload.playerName, payload.answer || '', Boolean(payload.isHost), now
      ];
      if (foundRow > 0) {
        sheet.getRange(foundRow, 1, 1, 9).setValues([rowValues]);
      } else {
        sheet.appendRow(rowValues);
      }
      return jsonResponse({ ok: true });
    }

    if (action === 'hostUpdate') {
      const now = new Date().toISOString();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).toUpperCase() === roomId) {
          if (payload.round !== undefined) sheet.getRange(i + 1, 2).setValue(payload.round);
          if (payload.status !== undefined) sheet.getRange(i + 1, 3).setValue(payload.status);
          if (payload.question !== undefined) sheet.getRange(i + 1, 4).setValue(payload.question);
          if (payload.clearAnswers) sheet.getRange(i + 1, 7).setValue('');
          sheet.getRange(i + 1, 9).setValue(now);
        }
      }
      return jsonResponse({ ok: true });
    }

    if (action === 'kickPlayer') {
      for (let i = data.length - 1; i >= 1; i--) {
        if (String(data[i][0]).toUpperCase() === roomId && String(data[i][4]) === String(payload.targetPlayerId)) {
          sheet.deleteRow(i + 1);
        }
      }
      return jsonResponse({ ok: true });
    }

    return jsonResponse({ ok: false, error: 'Unknown action' });
  } finally {
    lock.releaseLock();
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
