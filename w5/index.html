Google Apps Script 後端架構 (TelepathyData)

支援多房間分流與多列資料管理 (doGet / doPost)

試算表欄位設計：系統會自動建立名為 TelepathyData 的工作表，並管理以下 9 個欄位：
roomId | playerId | playerName | role | topic | answer | status | round | updatedAt 
const SHEET_NAME = 'TelepathyData';

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow([
      'roomId', 'playerId', 'playerName', 'role', 
      'topic', 'answer', 'status', 'round', 'updatedAt'
    ]);
  }
  return sheet;
}

// 讀取指定房間的所有玩家列資料
function doGet(e) {
  const roomId = (e.parameter.roomId || '').toUpperCase().trim();
  const sheet = getOrCreateSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = [];

  for (let i = 1; i < data.length; i++) {
    const rowRoom = String(data[i][0] || '').toUpperCase().trim();
    if (!roomId || rowRoom === roomId) {
      let obj = {};
      headers.forEach((h, idx) => obj[h] = data[i][idx]);
      rows.push(obj);
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, rows: rows }))
    .setMimeType(ContentService.MimeType.JSON);
}

// 寫入/更新玩家作答、房主出題、全體開牌或下一回合
function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action; // 'upsertPlayer' | 'updateRoomState' | 'nextRound'
    const roomId = String(payload.roomId || '').toUpperCase().trim();
    const sheet = getOrCreateSheet();
    const data = sheet.getDataRange().getValues();
    const now = new Date().toISOString();

    if (action === 'upsertPlayer') {
      const pId = String(payload.playerId);
      let foundRow = -1;
      let roomTopic = payload.topic || '說到夏天會想到的第一種水果？';
      let roomStatus = 'waiting';
      let roomRound = 1;

      // 取得該房間目前的統一題目與狀態
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).toUpperCase() === roomId) {
          if (data[i][4]) roomTopic = data[i][4];
          if (data[i][6]) roomStatus = data[i][6];
          if (data[i][7]) roomRound = data[i][7];
          if (String(data[i][1]) === pId) foundRow = i + 1;
        }
      }

      const finalTopic = payload.isHost && payload.topic ? payload.topic : roomTopic;
      const rowValues = [
        roomId,
        pId,
        payload.playerName || '匿名玩家',
        payload.role || 'player',
        finalTopic,
        payload.answer !== undefined ? payload.answer : '',
        roomStatus,
        roomRound,
        now
      ];

      if (foundRow > 0) {
        sheet.getRange(foundRow, 1, 1, 9).setValues([rowValues]);
      } else {
        sheet.appendRow(rowValues);
      }
    } 
    else if (action === 'updateRoomState') {
      // 房主更新題目或觸發全體開牌 (status = 'revealed')
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).toUpperCase() === roomId) {
          if (payload.topic !== undefined) sheet.getRange(i + 1, 5).setValue(payload.topic);
          if (payload.status !== undefined) sheet.getRange(i + 1, 7).setValue(payload.status);
          sheet.getRange(i + 1, 9).setValue(now);
        }
      }
    } 
    else if (action === 'nextRound') {
      // 清空全房間答案並回合 +1
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).toUpperCase() === roomId) {
          const nextR = (Number(data[i][7]) || 1) + 1;
          if (payload.topic !== undefined) sheet.getRange(i + 1, 5).setValue(payload.topic);
          sheet.getRange(i + 1, 6).setValue('');
          sheet.getRange(i + 1, 7).setValue('waiting');
          sheet.getRange(i + 1, 8).setValue(nextR);
          sheet.getRange(i + 1, 9).setValue(now);
        }
      }
    }

    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
